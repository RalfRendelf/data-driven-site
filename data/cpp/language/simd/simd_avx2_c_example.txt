#include <immintrin.h>
#include <stdio.h>

int main() {
    __m256i a = _mm256_set_epi32(1,2,3,4,5,6,7,8);
    __m256i b = _mm256_set_epi32(8,7,6,5,4,3,2,1);
    __m256i c = _mm256_mullo_epi32(a, b);

    int result[8];
    _mm256_storeu_si256((__m256i*)result, c);

    for (int i = 0; i < 8; i++)
        printf("%d ", result[i]);
}
