#include <emmintrin.h>
#include <stdio.h>

int main() {
    __m128i a = _mm_set_epi32(1, 2, 3, 4);
    __m128i b = _mm_set_epi32(5, 6, 7, 8);
    __m128i c = _mm_add_epi32(a, b);

    int result[4];
    _mm_storeu_si128((__m128i*)result, c);

    for (int i = 0; i < 4; i++)
        printf("%d ", result[i]);
}
