#include <immintrin.h>
#include <stdio.h>

int main() {
    __m256 a = _mm256_set_ps(1,2,3,4,5,6,7,8);
    __m256 b = _mm256_set_ps(8,7,6,5,4,3,2,1);
    __m256 c = _mm256_add_ps(a, b);

    float result[8];
    _mm256_storeu_ps(result, c);

    for (int i = 0; i < 8; i++)
        printf("%.1f ", result[i]);
}
