#include <boost/simd/pack.hpp>
#include <iostream>

using namespace boost::simd;

int main() {
    pack<int, 4> values{1, 2, 3, 4};
    pack<int, 4> mask{0, 1, 0, 1};

    pack<int, 4> result = if_else(mask, values * 10, values);

    for (int i = 0; i < result.size(); ++i)
        std::cout << result[i] << " ";
}
